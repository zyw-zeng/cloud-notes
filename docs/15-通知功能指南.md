# 通知系统 API 使用指南

## 🚀 快速开始

通知系统已完成开发，支持 **WebSocket 实时推送** 和 **RESTful API 轮询** 两种方式。

## 📡 WebSocket 实时推送（推荐）

### 前端连接 WebSocket

```javascript
// 1. 引入 SockJS 和 STOMP 客户端
import SockJS from 'sockjs-client';
import { Client } from '@stomp/stompjs';

// 2. 创建连接
const socket = new SockJS('http://localhost:8080/ws');
const stompClient = new Client({
    webSocketFactory: () => socket,
    reconnectDelay: 5000,
    heartbeatIncoming: 4000,
    heartbeatOutgoing: 4000
});

// 3. 连接成功后订阅通知
stompClient.onConnect = (frame) => {
    console.log('WebSocket 连接成功');
    
    // 订阅用户的私有通知队列
    stompClient.subscribe('/user/queue/notifications', (message) => {
        const notification = JSON.parse(message.body);
        console.log('收到新通知:', notification);
        
        // 显示通知（可以使用 Element UI 的 Notification 组件）
        showNotification(notification);
    });
};

// 4. 连接错误处理
stompClient.onStompError = (frame) => {
    console.error('WebSocket 错误:', frame);
};

// 5. 激活连接
stompClient.activate();

// 显示通知的函数
function showNotification(notification) {
    // 使用 Element UI
    this.$notify({
        title: notification.title,
        message: notification.content,
        type: 'info',
        duration: 5000,
        onClick: () => {
            // 点击通知跳转到相关页面
            if (notification.relatedType === 'TODO') {
                this.$router.push(`/todos/${notification.relatedId}`);
            }
        }
    });
    
    // 更新未读通知数量
    this.refreshUnreadCount();
}
```

## 📋 REST API 接口

### 1. 获取所有通知
```
GET /api/notifications
```

**响应示例**：
```json
{
  "code": 200,
  "message": "操作成功",
  "data": [
    {
      "id": 1,
      "title": "任务提醒",
      "content": "任务提醒：完成项目文档\n截止时间：2025-11-30T18:00:00",
      "type": "TODO_REMINDER",
      "relatedId": 5,
      "relatedType": "TODO",
      "isRead": false,
      "createdAt": "2025-11-30T17:00:00"
    }
  ]
}
```

---

### 2. 获取未读通知
```
GET /api/notifications/unread
```

---

### 3. 获取最近的通知
```
GET /api/notifications/recent?limit=10
```

**参数**：
- `limit`: 返回的通知数量（默认10条）

---

### 4. 获取未读通知数量
```
GET /api/notifications/unread/count
```

**响应示例**：
```json
{
  "code": 200,
  "message": "操作成功",
  "data": 5
}
```

---

### 5. 标记通知为已读
```
POST /api/notifications/{id}/read
```

---

### 6. 标记所有通知为已读
```
POST /api/notifications/read-all
```

---

### 7. 清理旧的已读通知
```
DELETE /api/notifications/clean
```

清理 30 天前的已读通知。

## 🎯 前端集成示例（Vue 3 + Element Plus）

### 通知中心组件

```vue
<template>
  <div class="notification-center">
    <!-- 通知图标 -->
    <el-badge :value="unreadCount" :hidden="unreadCount === 0">
      <el-button icon="Bell" @click="showDrawer = true" circle />
    </el-badge>

    <!-- 通知抽屉 -->
    <el-drawer v-model="showDrawer" title="通知中心" size="400px">
      <div class="notification-actions">
        <el-button size="small" @click="markAllAsRead">全部已读</el-button>
        <el-button size="small" @click="cleanOld">清理已读</el-button>
      </div>

      <el-scrollbar height="calc(100vh - 150px)">
        <div
          v-for="notification in notifications"
          :key="notification.id"
          :class="['notification-item', { unread: !notification.isRead }]"
          @click="handleNotificationClick(notification)"
        >
          <div class="notification-title">{{ notification.title }}</div>
          <div class="notification-content">{{ notification.content }}</div>
          <div class="notification-time">
            {{ formatTime(notification.createdAt) }}
          </div>
        </div>
      </el-scrollbar>
    </el-drawer>
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';
import { Client } from '@stomp/stompjs';
import SockJS from 'sockjs-client';
import { ElNotification } from 'element-plus';
import axios from 'axios';

const showDrawer = ref(false);
const notifications = ref([]);
const unreadCount = ref(0);
let stompClient = null;

// 初始化 WebSocket
onMounted(() => {
  connectWebSocket();
  loadNotifications();
  loadUnreadCount();
});

// 连接 WebSocket
function connectWebSocket() {
  const socket = new SockJS('http://localhost:8080/ws');
  stompClient = new Client({
    webSocketFactory: () => socket
  });

  stompClient.onConnect = () => {
    console.log('WebSocket 连接成功');
    
    stompClient.subscribe('/user/queue/notifications', (message) => {
      const notification = JSON.parse(message.body);
      
      // 添加到通知列表
      notifications.value.unshift(notification);
      unreadCount.value++;
      
      // 显示桌面通知
      ElNotification({
        title: notification.title,
        message: notification.content,
        type: 'info',
        onClick: () => handleNotificationClick(notification)
      });
    });
  };

  stompClient.activate();
}

// 加载通知列表
async function loadNotifications() {
  const { data } = await axios.get('/api/notifications/recent?limit=20');
  notifications.value = data.data;
}

// 加载未读数量
async function loadUnreadCount() {
  const { data } = await axios.get('/api/notifications/unread/count');
  unreadCount.value = data.data;
}

// 标记所有为已读
async function markAllAsRead() {
  await axios.patch('/api/notifications/read-all');
  notifications.value.forEach(n => n.isRead = true);
  unreadCount.value = 0;
  ElNotification.success('已全部标记为已读');
}

// 清理已读通知
async function cleanOld() {
  await axios.delete('/api/notifications/clean');
  await loadNotifications();
  ElNotification.success('已清理旧通知');
}

// 点击通知
async function handleNotificationClick(notification) {
  if (!notification.isRead) {
    await axios.patch(`/api/notifications/${notification.id}/read`);
    notification.isRead = true;
    unreadCount.value--;
  }
  
  // 跳转到相关页面
  if (notification.relatedType === 'TODO') {
    // router.push(`/todos/${notification.relatedId}`);
  }
}

// 断开连接
onUnmounted(() => {
  if (stompClient) {
    stompClient.deactivate();
  }
});
</script>
```

## 📊 数据库表结构

```sql
CREATE TABLE notifications (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT NOT NULL,
    title VARCHAR(255) NOT NULL,
    content TEXT,
    type VARCHAR(50) NOT NULL,
    related_id BIGINT,
    related_type VARCHAR(50),
    is_read BOOLEAN NOT NULL DEFAULT FALSE,
    read_at DATETIME,
    created_at DATETIME NOT NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_read (user_id, is_read),
    INDEX idx_created_at (created_at)
);
```
